export function UserDocCards() {
  const { metadata } = useDoc();
  const items = useCurrentSidebarSiblings().filter((item) => item.docId !== metadata.id);
  return <DocCardList items={items} />;
}

This section contains documentation intended for end users of Budgie Desktop 10.10.x, the currently supported release. Budgie Desktop 10.9.x and earlier are no longer supported, and Budgie 11 is still in development. You can check which version you have installed by running `budgie-desktop --version`.

New to Budgie? See [Get Budgie](https://buddiesofbudgie.org/get-budgie) to find a distribution that ships it.
